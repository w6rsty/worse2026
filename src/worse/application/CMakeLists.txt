set(APPLICATION_MODULES
    window.cppm
)
list(TRANSFORM APPLICATION_MODULES PREPEND ${CMAKE_CURRENT_SOURCE_DIR}/)

file(GLOB_RECURSE APPLICATION_HEADERS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/*.hpp)
file(GLOB_RECURSE APPLICATION_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp)

worse_add_library(application
    MODULES ${APPLICATION_MODULES}
    HEADERS ${APPLICATION_HEADERS}
    SOURCES ${APPLICATION_SOURCES}
    PRIVATE_DEPS
        worse::core
        glfw
)
